# vinboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> **Committing:** The repo owner's standing instruction is *the agent does not stage or commit*. Every "Commit" step below marks a checkpoint where **the user** reviews and commits. Do not run `git add`/`git commit` yourself unless the user says so.

**Goal:** A self-hosted single-user Pinboard replacement — one self-contained Zig binary with embedded SQLite, exposing a clean REST API + htmx/Tailwind web UI, page archiving via SingleFile, plus an Emacs client `vinboard.el`. Deployed on `hotter.myaddr.dev` behind Caddy.

**Architecture:** One Zig binary run under `dtach`, holding one SQLite file. Caddy reverse-proxies `/vinboard` → `localhost:4670` and applies the existing `/bookmarks` token+cookie gate (app has no auth). Background worker thread archives pages by shelling out to `single-file`. `vinboard.el` is a thin REST client.

**Tech Stack:** Zig 0.15.2; `httpz` (karlseguin/http.zig) for the HTTP server; SQLite via the bundled C amalgamation + `@cImport`; htmx + Tailwind via CDN (no JS build); `single-file-cli` (headless Chrome) as the archiver; Emacs Lisp for the client.

**Reference docs to keep open:**
- httpz README: https://github.com/karlseguin/http.zig (router, `Request`, `Response`, `res.json`).
- SQLite C API: https://www.sqlite.org/c3ref/intro.html ; FTS5: https://www.sqlite.org/fts5.html
- Zig 0.15 std: `std.Thread`, `std.process.Child`, `std.json`.

**Conventions:**
- Defaults from the spec: db `~/vinboard/vinboard.db`, port `4670`, base-path `/vinboard`, reuse the `/bookmarks` Caddy token. Import deferred until a Pinboard JSON export exists on disk.
- Run all `zig` commands from the repo root `~/Developer/src/github.com/velppa/vinboard`.
- Test command throughout: `zig build test`.

pub const Db = struct {
    handle: *c.sqlite3,

    pub fn open(path: [:0]const u8) !Db {
        var h: ?*c.sqlite3 = null;
        if (c.sqlite3_open(path.ptr, &h) != c.SQLITE_OK) return error.OpenFailed;
        return .{ .handle = h.? };
    }

    pub fn openMemory() !Db {
        return open(":memory:");
    }

    pub fn close(self: *Db) void {
        _ = c.sqlite3_close(self.handle);
    }

    /// Run one or more statements with no result rows.
    pub fn exec(self: *Db, sql: [:0]const u8) !void {
        var errmsg: [*c]u8 = null;
        if (c.sqlite3_exec(self.handle, sql.ptr, null, null, &errmsg) != c.SQLITE_OK) {
            if (errmsg != null) {
                std.log.err("sqlite exec: {s}", .{errmsg});
                c.sqlite3_free(errmsg);
            }
            return error.ExecFailed;
        }
    }

    /// Last error message for logging.
    pub fn errMsg(self: *Db) [*:0]const u8 {
        return c.sqlite3_errmsg(self.handle);
    }
};

    pub fn bindText(self: *Stmt, i: c_int, v: []const u8) void {

    pub fn bindInt(self: *Stmt, i: c_int, v: i64) void {
        _ = c.sqlite3_bind_int64(self.ptr, i, v);

    pub fn bindNull(self: *Stmt, i: c_int) void {
        _ = c.sqlite3_bind_null(self.ptr, i);
    }
    /// Returns true if a row is available, false when done.
    pub fn step(self: *Stmt) !bool {
        return switch (c.sqlite3_step(self.ptr)) {
            c.SQLITE_ROW => true,
            c.SQLITE_DONE => false,
            else => error.StepFailed,
        };
    }
    pub fn columnInt(self: *Stmt, i: c_int) i64 {
        return c.sqlite3_column_int64(self.ptr, i);

    pub fn columnText(self: *Stmt, i: c_int) []const u8 {

        return @as([*]const u8, @ptrCast(p))[0..len];

    var out = std.ArrayList(u8){};

            if (matchTag(html, i, "script")) { i = skipBlock(html, i, "</script>"); continue; }
            if (matchTag(html, i, "style")) { i = skipBlock(html, i, "</style>"); continue; }
            // skip to '>'
            while (i < html.len and html[i] != '>') i += 1;
            if (i < html.len) i += 1;
            if (!last_space) { try out.append(alloc, ' '); last_space = true; }
            continue;
        }
        const ch = html[i];

        if (ch == ' ' or ch == '\n' or ch == '\t' or ch == '\r') {
            if (!last_space) { try out.append(alloc, ' '); last_space = true; }

pub const Worker = struct {
    app: *server.App,
    archiver_cmd: []const u8, // "single-file" in prod, fixture path in tests
    poll_ms: u64 = 2000,
    stop: bool = false,

    pub fn run(self: *Worker) void {
        while (!self.stop) {
            self.tick() catch |e| std.log.err("archive tick: {s}", .{@errorName(e)});
            std.Thread.sleep(self.poll_ms * std.time.ns_per_ms);
        }
    }

    /// Process one pending row, if any. Public so tests can call it directly.
    pub fn tick(self: *Worker) !void {
        const job = (try self.nextPending()) orelse return;
        defer self.app.gpa.free(job.url);
        const html = self.fetch(job.url) catch {
            try self.markFailed(job.id);
            return;
        };
        defer self.app.gpa.free(html);
        const text = try strip.toText(self.app.gpa, html);
        defer self.app.gpa.free(text);
        self.app.db_mutex.lock();
        defer self.app.db_mutex.unlock();
        try db_mod.setArchive(self.app.db, job.id, html, text, .done, std.time.timestamp());
    }

    const Job = struct { id: i64, url: []u8 };

    fn nextPending(self: *Worker) !?Job {
        self.app.db_mutex.lock();
        defer self.app.db_mutex.unlock();
        var q = try self.app.db.prepare(
            "SELECT a.bookmark_id, b.url FROM archive a JOIN bookmarks b ON b.id=a.bookmark_id WHERE a.status='pending' LIMIT 1;",
        );
        defer q.finalize();
        if (!try q.step()) return null;
        return .{ .id = q.columnInt(0), .url = try self.app.gpa.dupe(u8, q.columnText(1)) };
    }

    fn markFailed(self: *Worker, id: i64) !void {
        self.app.db_mutex.lock();
        defer self.app.db_mutex.unlock();
        try db_mod.setArchive(self.app.db, id, "", "", .failed, std.time.timestamp());
    }

    /// Run the archiver, capture stdout HTML. Caller frees.
    fn fetch(self: *Worker, url: []const u8) ![]u8 {
        var child = std.process.Child.init(&.{ self.archiver_cmd, url }, self.app.gpa);
        child.stdout_behavior = .Pipe;
        child.stderr_behavior = .Ignore;
        try child.spawn();
        const out = try child.stdout.?.readToEndAlloc(self.app.gpa, 32 * 1024 * 1024);
        errdefer self.app.gpa.free(out);
        const term = try child.wait();
        if (term != .Exited or term.Exited != 0) { self.app.gpa.free(out); return error.ArchiverFailed; }
        return out;
    }
};

;; Talks to a vinboard server over its REST API.
(require 'url)
(require 'json)

(defgroup vinboard nil "vinboard client." :group 'tools)
(defcustom vinboard-base-url "https://uiuo.nl/vinboard"
  "Base URL of the vinboard server (no trailing slash)."
  :type 'string)
(defcustom vinboard-token "uery3721lkjKd"
  "Caddy gate token appended as ?token=."
  :type 'string)

(defun vinboard--url (path &optional query)
  "Build URL for PATH with optional QUERY alist; always includes the gate token."
  (concat vinboard-base-url path "?token=" (url-hexify-string vinboard-token)
          (mapconcat (lambda (kv) (format "&%s=%s" (car kv)
                                          (url-hexify-string (format "%s" (cdr kv)))))
                     query "")))

(defun vinboard--request (method path &optional body query)
  "Send METHOD to PATH with optional BODY (alist→JSON) and QUERY. Return parsed JSON."
  (let* ((url-request-method method)
         (url-request-extra-headers '(("Content-Type" . "application/json")))
         (url-request-data (when body (encode-coding-string (json-encode body) 'utf-8))))
    (with-current-buffer (url-retrieve-synchronously (vinboard--url path query) t)
      (goto-char (point-min))
      (when (re-search-forward "\n\n" nil t)
        (let ((json-object-type 'alist) (json-array-type 'list))
          (ignore-errors (json-read)))))))

;;;###autoload

  "Add bookmark URL with optional TITLE and TAGS (space-separated string)."
  (interactive "sURL: \nsTitle: \nsTags: ")
  (let ((res (vinboard--request "POST" "/api/bookmarks"
              `((url . ,url) (title . ,(or title ""))
                (tags . ,(if (and tags (> (length tags) 0)) (split-string tags) []))))))
    (message "vinboard: added id %s" (alist-get 'id res))))
